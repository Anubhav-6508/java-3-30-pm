Question 1 – Static and Instance Initialization
class Test {
    static int a = 10;
    int b = 20;

    static {
        a = a++ + ++a;
    }

    {
        b = b++ + ++b;
    }

    Test() {
        a++;
        b++;
    }

    void display() {
        System.out.println(a + " " + b);
    }

    public static void main(String[] args) {
        Test t1 = new Test();
        Test t2 = new Test();

        t1.display();
        t2.display();
    }
}

Question 3 – Constructor Chaining with Instance Blocks
class Student {

    static int count = 0;
    int id = 100;

    {
        id += 10;
        count++;
        System.out.println("Instance Block: " + id);
    }

    Student() {
        this(50);
        System.out.println("Default Constructor");
    }

    Student(int id) {
        this.id = id;
        System.out.println("Parameterized Constructor: " + id);
    }

    public static void main(String[] args) {
        Student s1 = new Student();
        Student s2 = new Student(200);

        System.out.println(count);
    }
}

Question 6 – Constructor Calling Another Constructor
class Employee {

    static int count = 0;
    int salary = 1000;

    {
        count++;
        salary += 500;
    }

    Employee() {
        this(2000);
        salary += 100;
    }

    Employee(int salary) {
        this.salary += salary;
        count++;
    }

    Employee(String name) {
        this();
        salary += 300;
    }

    public static void main(String[] args) {

        Employee e1 = new Employee();
        Employee e2 = new Employee(5000);
        Employee e3 = new Employee("Java");

        System.out.println(count);
        System.out.println(e1.salary);
        System.out.println(e2.salary);
        System.out.println(e3.salary);
    }
}

Question 11 – Static Variable and Local Variable Shadowing

    {
        int x = 100;

Question 18 – Master Challenge: Everything Combined
class Parent {

    static int a = 10;
    int b = 20;

    static {
        a += 5;
    }

    {
        b += a;
    }

    Parent() {
        a++;
        b++;
    }

    Parent(int n) {
        this();
        b += n;
    }
}

class Child extends Parent {

    static int a = 100;
    int b = 200;

    static {
        a += Parent.a;
    }

    {
        b += a;
    }

    Child() {
        super(10);
        a++;
        b++;
    }

    Child(int n) {
        this();
        b += n;
        a += n;
    }

    public static void main(String[] args) {

        Child c1 = new Child();
        Child c2 = new Child(50);

        System.out.println(Parent.a);
        System.out.println(Child.a);

        System.out.println(c1.b);
        System.out.println(c2.b);

        System.out.println(c1 instanceof Parent);
    }
}
